# Google Login Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let customers sign in to Cardora with their Google account, in addition to email/password, using Google Identity Services ID-token verification.

**Architecture:** The frontend renders Google's official Sign-In button (`@react-oauth/google`), which returns a signed ID token directly to the browser — no redirect, no client secret. The frontend POSTs that token to a new `POST /auth/google` backend endpoint, which verifies it against Google's public keys (`google-auth`), finds-or-links-or-creates the matching `User` row, and returns the same JWT shape the existing `/auth/login` endpoint returns. The frontend stores that JWT exactly like a normal login.

**Tech Stack:** FastAPI, SQLAlchemy, Alembic, `google-auth` (Python), Next.js, `@react-oauth/google`, existing Axios client.

## Global Constraints

- Google Client ID (already created): `302204730669-7ut65trrtl5cl9tdc4lkmrspjs74e960.apps.googleusercontent.com`
- Backend env var name: `GOOGLE_CLIENT_ID`
- Frontend env var name: `NEXT_PUBLIC_GOOGLE_CLIENT_ID`
- Account linking: if a Google sign-in's email matches an existing email/password account, auto-link (set `google_id` on that row) rather than blocking.
- Out of scope for this plan: "set a password" UI for Google-only accounts, full OAuth Authorization Code flow, any other social provider, production CORS/publishing changes (tracked separately, see spec's "Deployment Notes").
- All new backend code follows the existing feature-module layout under `app/features/auth/`.
- All new frontend code follows the existing feature-module layout under `frontend/src/features/auth/`, and uses the shared Axios client (`@/shared/lib/api/client`) — never raw `fetch()`.
- Money/currency, multi-tenant scoping, and other Cardora-wide rules are unaffected by this feature (no monetary or cross-user data involved).

```python
"""
Google Sign-In — ID token verification and user resolution.

Google Identity Services hands the frontend a signed ID token (a JWT).
verify_google_token() checks that signature against Google's public keys
and confirms the token was issued for our app (the `aud` claim).
get_or_create_google_user() then maps the verified claims onto a User row.
"""

from google.oauth2 import id_token as google_id_token
from google.auth.transport import requests as google_requests
from sqlalchemy.orm import Session

from app.core.config import settings
from app.features.auth.models import User


def verify_google_token(token: str) -> dict:
    """Verify a Google ID token and return its claims. Raises ValueError if invalid."""
    return google_id_token.verify_oauth2_token(
        token,
        google_requests.Request(),
        settings.GOOGLE_CLIENT_ID,
    )


def get_or_create_google_user(db: Session, claims: dict) -> User:
    """
    Resolve a verified Google claims dict to a User row.

    - Existing google_id -> return that user (repeat Google login)
    - No google_id match but matching email -> link google_id onto that account
    - No match at all -> create a new Google-only user (no password)
    """
    google_id = claims["sub"]

    user = db.query(User).filter(User.google_id == google_id).first()
    if user:
        return user

    email = claims["email"]
    user = db.query(User).filter(User.email == email).first()
    if user:
        user.google_id = google_id
        db.commit()
        db.refresh(user)
        return user

    new_user = User(
        email=email,
        google_id=google_id,
        full_name=claims.get("name"),
        hashed_password=None,
        timezone="Asia/Singapore",
        is_active=True,
        is_verified=True,
    )
    db.add(new_user)
    db.commit()
    db.refresh(new_user)
    return new_user
```

@auth_router.post(
    "/google",
    response_model=Token,
    status_code=status.HTTP_200_OK,
    summary="Sign in or register with Google"
)
async def google_login(
    request: GoogleLoginRequest,
    db: Session = Depends(get_db)
):
    """Verify a Google ID token and issue our own JWT for the matching user."""
    try:
        claims = verify_google_token(request.id_token)
    except ValueError:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid Google token",
        )

    if not claims.get("email_verified", False):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Google account email not verified",
        )

    user = get_or_create_google_user(db, claims)

    if not user.is_active:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="User account is inactive"
        )

    access_token = create_access_token(
        data={"sub": str(user.id), "email": user.email}
    )

    return Token(access_token=access_token, token_type="bearer")
```

"use client";

import { useRouter } from "next/navigation";
import { GoogleLogin, CredentialResponse } from "@react-oauth/google";
import { googleLogin } from "../api/authApi";
import { extractErrorMessage } from "@/shared/lib/utils/errorHandler";

interface GoogleSignInButtonProps {
  onError: (message: string) => void;
}

export default function GoogleSignInButton({ onError }: GoogleSignInButtonProps) {
  const router = useRouter();

  const handleSuccess = async (credentialResponse: CredentialResponse) => {
    if (!credentialResponse.credential) {
      onError("Google sign-in failed, please try again");
      return;
    }

    try {
      await googleLogin(credentialResponse.credential);

      const redirectPath = localStorage.getItem("redirectAfterLogin");
      if (redirectPath) {
        localStorage.removeItem("redirectAfterLogin");
        router.push(redirectPath);
      } else {
        router.push("/dashboard");
      }
    } catch (err) {
      onError(extractErrorMessage(err, "Google sign-in failed, please try again"));
    }
  };

  return (
    <div className="flex justify-center">
      <GoogleLogin
        onSuccess={handleSuccess}
        onError={() => onError("Google sign-in failed, please try again")}
        width="384"
      />
    </div>
  );
}
```
